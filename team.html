${player.number || "-"}

${player.name || "Unnamed Player"}

${player.position || "Player"}

${startingBadge} ${captainBadge}
`; return card; } // =============================== // LOAD SQUAD // =============================== async function loadSquad() { if (!selectedTeam) { teamName.textContent = "No Team Selected"; return; } try { const playersSnapshot = await getDocs(collection(db, "players")); const goalkeepers = document.getElementById("goalkeepers"); const defenders = document.getElementById("defenders"); const midfielders = document.getElementById("midfielders"); const forwards = document.getElementById("forwards"); goalkeepers.innerHTML = ""; defenders.innerHTML = ""; midfielders.innerHTML = ""; forwards.innerHTML = ""; let foundPlayers = false; playersSnapshot.forEach((doc) => { const player = doc.data(); // Only players from selected team if ( String(player.team).trim().toLowerCase() !== String(selectedTeam).trim().toLowerCase() ) { return; } foundPlayers = true; const card = createPlayerCard(player); const position = String(player.position || "") .trim() .toLowerCase(); // =============================== // POSITION // =============================== if ( position.includes("goal") || position.includes("keeper") || position === "gk" ) { goalkeepers.appendChild(card); } else if ( position.includes("def") || position === "cb" || position === "lb" || position === "rb" ) { defenders.appendChild(card); } else if ( position.includes("mid") || position === "cm" || position === "dm" || position === "am" ) { midfielders.appendChild(card); } else if ( position.includes("for") || position.includes("attack") || position === "fw" || position === "st" ) { forwards.appendChild(card); } else { midfielders.appendChild(card); } }); // =============================== // NO PLAYERS // =============================== if (!foundPlayers) { const message = `
No players registered for this team yet.
`; goalkeepers.innerHTML = message; } } catch (error) { console.error("Error loading squad:", error); teamName.textContent = "Unable to load squad"; } } // =============================== // START // =============================== loadSquad();
